import numpy as np
from scipy import ndimage
from llm_mrg import blocks as B

DESCRIPTION = ("Spectral-angle region growing: all bands normalised to unit length so Euclidean "
               "distance tracks spectral angle, grown from seeds in order of local spectral "
               "homogeneity under a seed-referenced threshold, within a 3x3 grid; pixel vote of an "
               "RBF SVM on PCs, smoothed PCs and local PC std.")
PARAMS = {"threshold": 0.12, "grad_sigma": 1.0, "grid": 3,
          "ctx_sigma1": 1.0, "ctx_sigma2": 4.0, "std_size": 7}
CLASSIFIER = {"kernel": "rbf", "C": 300.0, "gamma": "scale"}
LABELING = "pixel_vote"


def segment(ctx, params):
    x = np.asarray(ctx.bands, dtype=np.float64)
    x = x - x.min(axis=(0, 1), keepdims=True)
    x = x / (np.linalg.norm(x, axis=2, keepdims=True) + 1e-12)
    p = np.asarray(ctx.pcs[..., :3], dtype=np.float64)
    g = sum(ndimage.gaussian_gradient_magnitude(p[..., i], params["grad_sigma"]) for i in range(3))
    order = np.argsort(g.ravel(), kind="stable")
    cells = B.grid_cells(ctx.H, ctx.W, params["grid"])
    return B.grow(ctx, x, order=order, threshold=params["threshold"], criterion="seed", cells=cells)


def pixel_features(ctx, params):
    p = np.asarray(ctx.pcs, dtype=np.float32)
    k = p.shape[2]
    s1 = np.stack([ndimage.gaussian_filter(p[..., i], params["ctx_sigma1"]) for i in range(k)], -1)
    s2 = np.stack([ndimage.gaussian_filter(p[..., i], params["ctx_sigma2"]) for i in range(k)], -1)
    sz = params["std_size"]
    mu = np.stack([ndimage.uniform_filter(p[..., i], sz) for i in range(k)], -1)
    sq = np.stack([ndimage.uniform_filter(p[..., i] ** 2, sz) for i in range(k)], -1)
    sd = np.sqrt(np.maximum(sq - mu ** 2, 0))
    return np.concatenate([p, s1, s2, sd], -1)
