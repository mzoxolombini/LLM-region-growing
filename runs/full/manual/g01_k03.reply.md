"""M1 on D0: grouped accuracy (78-89) is the weakest term; region-level texture on tiny regions is
weak evidence. Keep the MRG regions, but classify pixels from their PCs and a local context
(smoothed PCs) with an RBF SVM and let each MRG region take the majority vote."""
import numpy as np
from scipy import ndimage
from llm_mrg import blocks as B

DESCRIPTION = ("Baseline MRG regions with pixel-vote labelling: an RBF SVM classifies pixels from "
               "PCs and 2-pixel smoothed PCs, each MRG region takes the majority vote.")
PARAMS = {"t_intensity": 0.05, "t_orientation": 0.7853981633974483, "n_bins": 256, "grid": 3,
          "connectivity": 8, "ctx_sigma": 2.0}
CLASSIFIER = {"kernel": "rbf", "C": 100.0, "gamma": "scale"}
LABELING = "pixel_vote"


def segment(ctx, params):
    return B.mrg(ctx, t_intensity=params["t_intensity"], t_orientation=params["t_orientation"],
                 n_bins=params["n_bins"], grid=params["grid"], connectivity=params["connectivity"])


def pixel_features(ctx, params):
    p = np.asarray(ctx.pcs, dtype=np.float32)
    s = np.stack([ndimage.gaussian_filter(p[..., i], params["ctx_sigma"]) for i in range(p.shape[2])], -1)
    return np.concatenate([p, s], -1)
