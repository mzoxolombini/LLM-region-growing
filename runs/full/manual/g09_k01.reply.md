import numpy as np
from scipy import ndimage
from sklearn.cluster import MiniBatchKMeans
from llm_mrg import blocks as B

DESCRIPTION = ("Spectral-angle clustering: k-means on unit-normalised spectra (cosine geometry), "
               "connected components of the cluster map with singletons merged into their closest "
               "neighbour; pixel vote of an RBF SVM on PCs, two smoothing scales and local PC std.")
PARAMS = {"k": 60, "sample": 40000, "min_size": 2, "seed": 0,
          "ctx_sigma1": 1.0, "ctx_sigma2": 4.0, "std_size": 7}
CLASSIFIER = {"kernel": "rbf", "C": 300.0, "gamma": "scale"}
LABELING = "pixel_vote"


def segment(ctx, params):
    x = np.asarray(ctx.bands, dtype=np.float32).reshape(-1, ctx.B)
    x = x - x.min(0, keepdims=True)
    x /= np.linalg.norm(x, axis=1, keepdims=True) + 1e-12
    idx = ctx.rng.choice(x.shape[0], size=min(params["sample"], x.shape[0]), replace=False)
    km = MiniBatchKMeans(n_clusters=params["k"], random_state=params["seed"], n_init=3).fit(x[idx])
    cl = km.predict(x).reshape(ctx.H, ctx.W)
    lab = B.connected_components(cl, 8, ctx)
    return B.merge_small(ctx, lab, ctx.pcs, min_size=params["min_size"])


def pixel_features(ctx, params):
    p = np.asarray(ctx.pcs, dtype=np.float32)
    k = p.shape[2]
    s1 = np.stack([ndimage.gaussian_filter(p[..., i], params["ctx_sigma1"]) for i in range(k)], -1)
    s2 = np.stack([ndimage.gaussian_filter(p[..., i], params["ctx_sigma2"]) for i in range(k)], -1)
    sz = params["std_size"]
    mu = np.stack([ndimage.uniform_filter(p[..., i], sz) for i in range(k)], -1)
    sq = np.stack([ndimage.uniform_filter(p[..., i] ** 2, sz) for i in range(k)], -1)
    return np.concatenate([p, s1, s2, np.sqrt(np.maximum(sq - mu ** 2, 0))], -1)
