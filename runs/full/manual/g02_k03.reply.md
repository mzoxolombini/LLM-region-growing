"""M1 on g01_01. Weakest point: Indian Pines (F 86.2, OA_grp 79.8, mF1 0.83) where lattice
markers create 25-pixel regions that cross field borders. Markers now come from regional minima
of the smoothed gradient, so regions follow edges, and regions also carry smoothed-PC context."""
import numpy as np
from scipy import ndimage
from skimage.segmentation import watershed
from skimage.morphology import h_minima
from llm_mrg import blocks as B

DESCRIPTION = ("Watershed from h-minima markers of the PC gradient so region borders follow spectral "
               "edges; regions described by PC mean/std, mean spectrum and smoothed-PC context, RBF SVM.")
PARAMS = {"n_pc": 5, "grad_sigma": 0.8, "h": 0.05, "ctx_sigma": 2.0}
CLASSIFIER = {"kernel": "rbf", "C": 100.0, "gamma": "scale"}
LABELING = "region"


def segment(ctx, params):
    p = np.asarray(ctx.pcs[..., : params["n_pc"]], dtype=np.float64)
    g = sum(ndimage.gaussian_gradient_magnitude(p[..., i], params["grad_sigma"]) for i in range(p.shape[2]))
    g = g / (g.max() + 1e-12)
    markers, _ = ndimage.label(h_minima(g, params["h"]))
    return B.relabel(watershed(g, markers) - 1)


def region_features(ctx, regions, rows, params):
    p = np.asarray(ctx.pcs, dtype=np.float32)
    s = np.stack([ndimage.gaussian_filter(p[..., i], params["ctx_sigma"]) for i in range(p.shape[2])], -1)
    return np.hstack([B.region_means(p, regions, rows), B.region_stds(p, regions, rows),
                      B.region_means(ctx.bands, regions, rows), B.region_means(s, regions, rows)])
