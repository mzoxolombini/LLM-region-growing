import numpy as np
from scipy import ndimage
from skimage.segmentation import watershed
from skimage.feature import peak_local_max
from llm_mrg import blocks as B

DESCRIPTION = ("Marker-controlled compact watershed on the summed gradient magnitude of the first "
               "five PCs; markers on a regular lattice; regions described by PC mean/std and mean "
               "spectrum, RBF SVM.")
PARAMS = {"n_pc": 5, "grad_sigma": 1.0, "marker_step": 5, "compactness": 0.01}
CLASSIFIER = {"kernel": "rbf", "C": 100.0, "gamma": "scale"}
LABELING = "region"


def segment(ctx, params):
    p = np.asarray(ctx.pcs[..., : params["n_pc"]], dtype=np.float64)
    g = sum(ndimage.gaussian_gradient_magnitude(p[..., i], params["grad_sigma"]) for i in range(p.shape[2]))
    st = params["marker_step"]
    markers = np.zeros(g.shape, np.int32)
    yy, xx = np.mgrid[st // 2: g.shape[0]: st, st // 2: g.shape[1]: st]
    markers[yy, xx] = np.arange(1, yy.size + 1).reshape(yy.shape)
    lab = watershed(g, markers, compactness=params["compactness"])
    return B.relabel(lab - 1)


def region_features(ctx, regions, rows, params):
    return np.hstack([B.region_means(ctx.pcs, regions, rows), B.region_stds(ctx.pcs, regions, rows),
                      B.region_means(ctx.bands, regions, rows)])
