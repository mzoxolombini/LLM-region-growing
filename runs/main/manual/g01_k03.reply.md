"""M1 refinement of the baseline MRG. Feedback: tiny regions everywhere (mean size 3-6). The
cause is the orientation test: in flat areas the Sobel orientation is noise, so neighbours of a
seed fail the test at random. Fix inside the MRG rule: orientation is computed on a smoothed V
channel and enforced only where both gradients are strong; the intensity threshold is set from
a robust noise estimate of V so it adapts to each sensor. Same seeds, same 3x3 grid."""
import numpy as np
from llm_mrg import blocks as B

DESCRIPTION = ("Baseline MRG with a noise-adaptive intensity threshold and an orientation "
               "constraint applied only where the smoothed gradient is strong, so homogeneous "
               "areas grow into larger regions; GLCM, PC means and mean spectrum describe regions.")

PARAMS = {
    "noise_factor": 3.0,
    "t_orientation": 0.7853981633974483,
    "grad_quantile": 0.6,
    "smooth_sigma": 1.0,
    "n_bins": 256,
    "grid": 3,
    "connectivity": 8,
    "glcm_levels": 16,
}

CLASSIFIER = {"kernel": "linear", "C": 1.0, "gamma": "scale"}
LABELING = "region"


def _noise_sigma(v):
    d = np.diff(v, axis=1).ravel()
    return 1.4826 * np.median(np.abs(d - np.median(d))) / np.sqrt(2.0)


def segment(ctx, params):
    v = np.asarray(ctx.v, dtype=np.float64)
    t_int = params["noise_factor"] * max(_noise_sigma(v), 1e-4)
    _, gmag = B.orientation_map(v, params["smooth_sigma"])
    min_grad = float(np.quantile(gmag, params["grad_quantile"]))
    return B.mrg(ctx, t_intensity=t_int, t_orientation=params["t_orientation"],
                 n_bins=params["n_bins"], grid=params["grid"], connectivity=params["connectivity"],
                 min_grad=min_grad, smooth_sigma=params["smooth_sigma"])


def region_features(ctx, regions, rows, params):
    g = B.glcm88(ctx, regions, rows, levels=params["glcm_levels"])
    pc = B.region_means(ctx.pcs, regions, rows)
    spec = B.region_means(ctx.bands, regions, rows)
    return np.hstack([g, pc, spec])
