"""M1 refinement of the baseline MRG. Feedback: regions average 3-6 pixels on every scene, so
88 GLCM features computed on 3-6 pixels are close to constant, and the grouped hold-out
(OA_grp 78 on Indian Pines and Houston) is the weakest term. The growth rule is kept; the
region description is replaced by first- and second-order statistics of the ten PCs, which
stay informative on very small regions, plus the mean spectrum."""
import numpy as np
from llm_mrg import blocks as B

DESCRIPTION = ("Baseline MRG growth rule unchanged; regions described by the mean and standard "
               "deviation of the ten standardised PCs, the mean V value and the mean spectrum "
               "instead of GLCM texture, which is degenerate on 3-6 pixel regions.")

PARAMS = {
    "t_intensity": 0.05,
    "t_orientation": 0.7853981633974483,
    "n_bins": 256,
    "grid": 3,
    "connectivity": 8,
}

CLASSIFIER = {"kernel": "linear", "C": 1.0, "gamma": "scale"}
LABELING = "region"


def segment(ctx, params):
    return B.mrg(ctx, t_intensity=params["t_intensity"], t_orientation=params["t_orientation"],
                 n_bins=params["n_bins"], grid=params["grid"],
                 connectivity=params["connectivity"])


def region_features(ctx, regions, rows, params):
    m = B.region_means(ctx.pcs, regions, rows)
    s = B.region_stds(ctx.pcs, regions, rows)
    v = B.region_means(ctx.v, regions, rows)
    spec = B.region_means(ctx.bands, regions, rows)
    return np.hstack([m, s, v, spec])
